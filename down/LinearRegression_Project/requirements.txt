# -----------------------------------------------------------------------------
# 一、本项目的运行环境
# -----------------------------------------------------------------------------
# Python 版本：>= 3.9（开发与测试使用的是 Python 3.14.7）
# 操作系统  ：Windows / macOS / Linux 均可（代码中没有依赖特定系统的写法）
#
# 说明：核心训练代码（dataset.py / model.py / train.py）只依赖 NumPy，
#       其余两个包分别用于画图和生成报告 PDF。
# -----------------------------------------------------------------------------


# ---------------------------- 核心依赖（训练必需） ----------------------------

# NumPy：整个项目的唯一数值计算库。
#   用途：矩阵运算、随机数生成、以及所有手推梯度的向量化实现。
#   注意：本项目刻意【没有】使用 scikit-learn / pandas，
#         线性回归的模型、损失函数、梯度、优化算法全部是从零手写的。
numpy>=1.26,<3.0

# ---------------------------- 画图依赖（生成 results/*.png 必需） ----------------------------
# 若只想跑训练、不生成图片，可以用：python code/train.py --no-figures

# Matplotlib：绘制 loss 下降曲线、学习率对比图、拟合效果图等。
#   注意：脚本里已把缓存目录改到项目内的 .mplcache/，
#         避免某些环境下用户目录无写权限导致的报错。
matplotlib>=3.8

# ---------------------------- 报告依赖（重新生成 report.pdf 时必需） ----------------------------

# ReportLab：用代码生成 report.pdf（排版文字、表格，并把 results/*.png 插图进去）。
#   注意：Python 3.14 需要 ReportLab 5.x（4.x 及更早版本未支持 3.14）。
#         如果你用的是 Python 3.9 ~ 3.13，也可以装 4.x 版本。
reportlab>=4.0
